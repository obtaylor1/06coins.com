    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";
        import { getAuth, signInWithCustomToken, signInAnonymously } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js";
        import { getFirestore, doc, onSnapshot, updateDoc } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";

        // Global variables provided by the Canvas environment
        const appId = typeof __app_id !== 'undefined' ? __app_id : 'default-app-id';
        const firebaseConfig = typeof __firebase_config !== 'undefined' ? JSON.parse(__firebase_config) : null;
        const initialAuthToken = typeof __initial_auth_token !== 'undefined' ? __initial_auth_token : null;

        let db = null;
        let auth = null;
        let inventoryDocRef = null;

        // Inventory constants (Admin controlled, but tracked in Firestore)
        const INVENTORY_DOC_ID = 'apa120th_coin_stock';
        const INITIAL_PHASE_STOCK = 406; // Controlled Scarcity start number

        const initFirebase = async () => {
            if (!firebaseConfig) {
                console.error("Firebase configuration is missing. Cannot initialize Firestore.");
                document.getElementById('stock-counter').textContent = 'Inventory Error';
                return;
            }
            try {
                const app = initializeApp(firebaseConfig);
                db = getFirestore(app);
                auth = getAuth(app);

                // Authenticate
                if (initialAuthToken) {
                    await signInWithCustomToken(auth, initialAuthToken);
                } else {
                    await signInAnonymously(auth);
                }

                const userId = auth.currentUser?.uid || 'anonymous';
                console.log("Firebase initialized. User ID:", userId);

                // Set up the inventory document path
                inventoryDocRef = doc(db, 'artifacts', appId, 'public', 'data', 'coin_inventory', INVENTORY_DOC_ID);

                // Start real-time listener for the inventory count
                onSnapshot(inventoryDocRef, (docSnap) => {
                    if (docSnap.exists()) {
                        const data = docSnap.data();
                        const currentStock = data.remaining_stock || 0;
                        updateUI(currentStock);
                    } else {
                        // Document doesn't exist, assume starting stock
                        updateUI(INITIAL_PHASE_STOCK);
                        console.warn("Inventory document not found. UI defaulted to initial stock.");
                        // An admin function would create this document here if it didn't exist.
                    }
                }, (error) => {
                    console.error("Error listening to inventory:", error);
                    document.getElementById('stock-counter').textContent = 'Live Counter Offline';
                });

            } catch (error) {
                console.error("Firebase Initialization Failed:", error);
                document.getElementById('stock-counter').textContent = 'Initialization Error';
            }
        };

        const updateUI = (currentStock) => {
            const counterElement = document.getElementById('stock-counter');
            const buyButtons = document.querySelectorAll('.buy-button');

            counterElement.textContent = currentStock;

            if (currentStock <= 0) {
                counterElement.classList.add('text-red-600');
                document.getElementById('scarcity-status').textContent = 'Phase I is SOLD OUT!';
                document.getElementById('scarcity-status').classList.remove('color-gold');
                document.getElementById('scarcity-status').classList.add('text-red-600');

                buyButtons.forEach(btn => {
                    btn.textContent = 'JOIN WAITLIST';
                    btn.disabled = true;
                    btn.classList.remove('bg-gold');
                    btn.classList.add('bg-gray-700');
                });
            } else {
                 counterElement.classList.remove('text-red-600');
                 document.getElementById('scarcity-status').textContent = 'Limited Edition - Only';
                 document.getElementById('scarcity-status').classList.add('color-gold');
                 document.getElementById('scarcity-status').classList.remove('text-red-600');

                 buyButtons.forEach(btn => {
                    btn.textContent = 'SECURE YOUR COIN NOW';
                    btn.disabled = false;
                    btn.classList.add('bg-gold');
                    btn.classList.remove('bg-gray-700');
                });
            }
        };

        // Simulated function to decrement stock upon successful payment (called via webhook in a real scenario)
        // For this prototype, we'll simulate the reduction here.
        window.simulatePurchaseAndDecrementStock = async () => {
            if (!db || !inventoryDocRef) {
                console.error("Database not ready.");
                return;
            }
            try {
                const stockElement = document.getElementById('stock-counter');
                const currentStock = parseInt(stockElement.textContent) || 0;

                if (currentStock > 0) {
                    await updateDoc(inventoryDocRef, {
                        remaining_stock: currentStock - 1
                    });
                    console.log("Stock successfully decremented.");
                    // In a real app, this happens via a backend webhook, not the front end.
                } else {
                    console.log("Stock already at zero.");
                }

            } catch (error) {
                console.error("Failed to update stock:", error);
            }
        };

        // --- ADMIN REPLENISH FUNCTION (FOR CONTROLLED SCARCITY) ---
        window.replenishStock = async () => {
            // NOTE: In a production site, this function would be protected by strong user authentication and role-based access control.
            // For the prototype, we use a simple confirmation modal.
            const newStock = INITIAL_PHASE_STOCK; // Resetting to the initial phase batch size

            const confirmation = window.prompt(`ADMIN ACTION: Type 'RESET' to replenish the inventory counter to ${newStock} (starting Phase II).`);

            if (confirmation === 'RESET' && db && inventoryDocRef) {
                 try {
                    await updateDoc(inventoryDocRef, {
                        remaining_stock: newStock
                    });
                    alert(`Inventory successfully reset to ${newStock}. Sales Phase II is now active!`);
                } catch (error) {
                    console.error("Failed to replenish stock:", error);
                    alert("Replenishment failed. Check console for details.");
                }
            } else if (confirmation !== 'RESET') {
                 alert("Replenishment cancelled.");
            }
        };


        window.onload = initFirebase;
    </script>
